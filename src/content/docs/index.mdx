{/*
	Page counts are read from the content directory at build time so the section
	cards below can't drift out of date as pages are added. Section overview pages
	(`index.md`) don't count as content.
*/}
export const pages = Object.keys(import.meta.glob('./**/*.{md,mdx}'));
export const count = (dir, noun) => {
	const n = pages.filter((p) => p.startsWith(`./${dir}/`) && !/\/index\.mdx?$/.test(p)).length;
	return `${n} ${noun}${n === 1 ? '' : 's'}`;
};
